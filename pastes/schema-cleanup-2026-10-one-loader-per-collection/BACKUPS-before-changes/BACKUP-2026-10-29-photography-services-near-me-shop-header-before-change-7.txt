     ALAN RANGER — PRODUCT-LIST H1 DEMOTE FIX (photography-services-near-me)
     Where to paste: Page Settings -> Advanced -> PAGE HEADER CODE
                     INJECTION  (this page only)
     What it does:  Squarespace renders every product title in the
                    storae grid as an <h1> (h1.ProductList-title),
                    giving ~23 H1s on this page. This script runs after
                    load and converts ONLY those product-title H1s into
                    <h2>, leaving the real page H1 as the single page H1.
     Safe:          Does NOT touch schema, links, styling, or the page H1.
                    Reversible - remove this block to revert.
     ============================================================ -->
<script>
(function () {
  function demoteProductH1s() {
    // Product titles render as h1.ProductList-title inside .ProductList-item.
    // The real page H1 has no .ProductList-title class, so it is left alone.
    var productH1s = document.querySelectorAll('h1.ProductList-title');
    productH1s.forEach(function (h1) {
      if (h1.dataset.arDemoted === '1') return;          // idempotent
      var h2 = document.createElement('h2');
      h2.className = h1.className;                        // keeps ProductList-title styling
      if (h1.getAttribute('style')) h2.setAttribute('style', h1.getAttribute('style'));
      h2.innerHTML = h1.innerHTML;
      h2.dataset.arDemoted = '1';
      h1.parentNode.replaceChild(h2, h1);
    });
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', demoteProductH1s);
  } else {
    demoteProductH1s();
  }
  window.addEventListener('load', demoteProductH1s);
  setTimeout(demoteProductH1s, 1200);
  setTimeout(demoteProductH1s, 3000);
})();
</script>

      block.style.whiteSpace = "nowrap";
      const titleEl = document.createElement("h2");
      titleEl.textContent = "TLDR";
      const bodyEl = document.createElement("p");
      bodyEl.textContent = summary;
      block.appendChild(titleEl);
      block.appendChild(bodyEl);
      if (host.firstChild) host.insertBefore(block, host.firstChild);
      else host.appendChild(block);
    }
    injectTldrBlock(summaryText);
  } catch (e) {
    console.warn("Landing TLDR signal injection failed:", e);
  }
})();
</script>
